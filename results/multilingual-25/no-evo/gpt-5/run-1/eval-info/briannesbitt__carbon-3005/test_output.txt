+     * (for example $step from IntervalStep trait or local callbacks/translators).
+     *
+     * Only core interval scalar fields and a normalized timezone are exported.
+     */
+    public function __serialize(): array
+    {
+        $data = [
+            'y' => $this->y,
+            'm' => $this->m,
+            'd' => $this->d,
+            'h' => $this->h,
+            'i' => $this->i,
+            's' => $this->s,
+            'f' => $this->f,
+            'invert' => $this->invert,
+        ];
+
+        if (isset($this->days)) {
+            $data['days'] = $this->days;
+        }
+
+        // Normalize timezone to a string to ensure safe serialization.
+        if (isset($this->timezoneSetting)) {
+            $data['timezoneSetting'] = match (true) {
+                ($this->timezoneSetting === null) => null,
+                \is_string($this->timezoneSetting) => $this->timezoneSetting,
+                ($this->timezoneSetting instanceof DateTimeZone) => $this->timezoneSetting->getName(),
+                default => CarbonTimeZone::instance($this->timezoneSetting)->getName(),
+            };
+        }
+
+        return $data;
+    }
