+        $tz = match (true) {
+            ($this->timezoneSetting === null) => null,

+            ($this->timezoneSetting instanceof DateTimeZone) => $this->timezoneSetting->getName(),
+            default => CarbonTimeZone::instance($this->timezoneSetting)->getName(),
+        };

+            $export["timezone"] = $tz;
